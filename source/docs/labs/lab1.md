#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Лабораторная работа №1: Введение в Python
Автор: Иван Пушкин
Дата: 2026-03-03
"""

def get_user_input():
    """Запрос и валидация данных пользователя"""
    while True:
        name = input("Введите ваше имя: ").strip()
        if name:
            break
        print("❌ Имя не может быть пустым!")
    
    while True:
        try:
            age = int(input("Введите ваш возраст: "))
            if 0 <= age <= 120:
                return name, age
            print("❌ Возраст должен быть от 0 до 120!")
        except ValueError:
            print("❌ Введите корректное число!")

def main():
    """Основная функция программы"""
    print("👋 Добро пожаловать!")
    name, age = get_user_input()
    
    if age < 18:
        print(f"Привет, {name}! Вам {age} лет.")
    elif age < 60:
        print(f"Здравствуйте, {name}! Вам {age} лет.")
    else:
        print(f"Уважаемый {name}, вам {age} лет!")

if __name__ == "__main__":
    main()
